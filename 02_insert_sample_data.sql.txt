-- STEP 3: GENERATE 100 ADDITIONAL EMPLOYEES
-- Employee IDs: 1031 through 1130

INSERT INTO ptk_employees (
    employee_id,
    employee_name,
    job_title,
    salary,
    department_id,
    hire_date
)
SELECT 1030 + n,
       'Employee ' || TO_CHAR(1030 + n),
       CASE MOD(n, 6)
           WHEN 0 THEN 'Database Developer'
           WHEN 1 THEN 'HR Officer'
           WHEN 2 THEN 'Accountant'
           WHEN 3 THEN 'Sales Executive'
           WHEN 4 THEN 'Data Analyst'
           ELSE 'Software Engineer'
       END,
       45000 + MOD(n * 1379, 65001),
       10 * (1 + MOD(n - 1, 30)),
       DATE '2019-01-01' + MOD(n * 17, 2800)
FROM (
    SELECT LEVEL AS n
    FROM dual
    CONNECT BY LEVEL <= 100
);
