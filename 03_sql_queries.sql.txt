-- Query 1: Employee and department report
SELECT e.employee_id,
       e.employee_name,
       e.job_title,
       e.salary,
       d.department_name,
       d.location
FROM ptk_employees e
JOIN ptk_departments d
  ON e.department_id = d.department_id
ORDER BY e.employee_id;


-- Query 2: Salary statistics by department
SELECT d.department_name,
       COUNT(e.employee_id) AS total_employees,
       ROUND(AVG(e.salary), 2) AS average_salary,
       MAX(e.salary) AS highest_salary,
       MIN(e.salary) AS lowest_salary
FROM ptk_departments d
LEFT JOIN ptk_employees e
  ON d.department_id = e.department_id
GROUP BY d.department_id, d.department_name
ORDER BY average_salary DESC NULLS LAST;


-- Query 3: Employees earning above company average
SELECT employee_id,
       employee_name,
       job_title,
       salary,
       department_id
FROM ptk_employees
WHERE salary > (
    SELECT AVG(salary)
    FROM ptk_employees
)
ORDER BY salary DESC;


-- Query 4: Highest-paid employee in each department
SELECT department_name,
       employee_id,
       employee_name,
       job_title,
       salary
FROM (
    SELECT d.department_name,
           e.employee_id,
           e.employee_name,
           e.job_title,
           e.salary,
           RANK() OVER (
               PARTITION BY e.department_id
               ORDER BY e.salary DESC
           ) AS salary_rank
    FROM ptk_employees e
    JOIN ptk_departments d
      ON e.department_id = d.department_id
)
WHERE salary_rank = 1
ORDER BY department_name;


-- Query 5: Employees earning above their department average
SELECT e.employee_id,
       e.employee_name,
       d.department_name,
       e.salary
FROM ptk_employees e
JOIN ptk_departments d
  ON e.department_id = d.department_id
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM ptk_employees e2
    WHERE e2.department_id = e.department_id
)
ORDER BY d.department_name, e.salary DESC;
